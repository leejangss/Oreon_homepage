<script>
(() => {
  if (window.__oreonCategoryNavigation) return;
  window.__oreonCategoryNavigation = true;
  let request = 0;
  function move(el, offset = 0) {
    window.scrollTo({top: Math.max(0, scrollY + el.getBoundingClientRect().top + offset - 105),
      behavior: matchMedia('(prefers-reduced-motion: reduce)').matches ? 'auto' : 'smooth'});
  }
  function filter(category, ticket, attempt = 0) {
    if (ticket !== request) return;
    const stage = document.querySelector('#w202610103e49cc0075aa3 .omc');
    if (!stage) {if (attempt < 40) setTimeout(() => filter(category, ticket, attempt + 1), 250); return}
    document.dispatchEvent(new CustomEvent('oreon:category-select',{detail:{category,source:'oreon-time-v09-iframe'}}));
    setTimeout(() => {if (ticket === request) move(stage)},150);
  }
  window.addEventListener('message', e => {
    const frame = document.getElementById('oreon-time-motion-v09');
    if (!frame || e.source !== frame.contentWindow || e.origin !== 'https://leejangss.github.io') return;
    const d = e.data;
    if (!d || d.namespace !== 'OREON_TIME_V09' || typeof d.category !== 'string') return;
    const cat = d.category.toUpperCase().replace(/^CAPABILITY$/,'PROFESSIONALISM');
    if (!['CHALLENGE','PROFESSIONALISM','CREATIVITY','OPENNESS'].includes(cat)) return;
    if (d.type === 'values' && Number.isFinite(d.offset) && d.offset >= 0 && d.offset < 30000) {request++; move(frame,d.offset)}
    else if (d.type === 'board') filter(cat,++request);
  });
})();
</script>

<script>
(() => {
  const mobile = matchMedia('(max-width:767px)');
  const source = 'https://leejangss.github.io/Oreon_homepage/widgets/Oreon%20Time/mobile/OREON_Time_Mobile_Contact_V1.html?v=1';
  window.addEventListener('message', event => {
    const frame = document.getElementById('oreon-mobile-contact-restored');
    if (!frame || event.source !== frame.contentWindow || event.origin !== new URL(source).origin) return;
    const data = event.data;
    if (data?.type === 'oreon-contact-height' && Number.isFinite(data.height) && data.height > 0 && data.height < 5000)
      frame.style.height = data.height + 'px';
  });
  function restoreContact() {
    const board = document.getElementById('w202610103e49cc0075aa3');
    if (!board || !mobile.matches || new URLSearchParams(location.search).get('bmode') === 'view') return;
    const id = 'oreon-mobile-contact-restored';
    const existing = document.getElementById(id);
    if (existing) return;
    const frame = document.createElement('iframe');
    frame.id = id;
    frame.title = 'OREON TIME GET IN TOUCH';
    frame.src = source;
    frame.scrolling = 'no';
    board.after(frame);

  }
  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded',restoreContact,{once:true});
  else restoreContact();
  window.addEventListener('load',restoreContact,{once:true});
  mobile.addEventListener('change',restoreContact);
})();
</script>
